function(add_hex_test TEST_NAME INPUT_FILE EXPECTED_REGEX)
    add_test(
        NAME ${TEST_NAME}
        COMMAND Lab3CPP "${INPUT_FILE}"
    )
    set_tests_properties(${TEST_NAME} PROPERTIES
        PASS_REGULAR_EXPRESSION "${EXPECTED_REGEX}"
    )
endfunction()
